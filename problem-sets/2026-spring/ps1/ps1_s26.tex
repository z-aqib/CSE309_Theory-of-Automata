\begin{enumerate}

\item Using his intuition that languages accepted by finite automata require remembering only a finite pattern, Sheikh Chilly would like to argue that the following language $L_n$ is not regular. \emph{Obviously, he argues, remembering multiples of a given integer $n$ requires unbounded memory.} For the alphabet $\Sigma = \{ 0, 1\}$, let the language $L_n$ be given by
\begin{equation*}  
L_n = \{w \in \Sigma^* \, | \, w \textrm{ is a binary number that is a multiple of } n \} \, ,
\end{equation*}  
for $n \geq 1$. Show that for all valid choices of $n$, the language $L_n$ is regular.


\item Convert the following NFA into a DFA.

\item Let $\Sigma = \{a,b\}$ and $L_k = \{ ww : w \in \Sigma^* \textrm{ and $w$ is of length $k$}\}$. Show that for each $k$, no DFA can recognize $L_k$ with fewer than $2^k$ states.

\item For languages $A$ and $B$, let the \emph{interleave} of $A$ and $B$ be the language
\begin{equation*}
\{w \,| \, w = a_1b_1 \ldots a_kb_k, \textrm{ where } a_1 \ldots a_k \in A \textrm{ and } b_1 \ldots b_k \in B, \textrm{ each } a_i, b_i \in \Sigma \}.
\end{equation*}
Show that the class of regular languages is closed under \emph{interleave}.

\item Let $L = \{1^n : n \textrm{ is composite}\}$. Show that $L$ is not regular.


\end{enumerate}
